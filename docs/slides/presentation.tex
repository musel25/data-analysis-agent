\section{Context: why measure this}
% ###########################################################################

\begin{frame}{Motivation}
\begin{itemize}
  \item AI agents have improved fast --- they now \textbf{write and run code in a loop}, unsupervised.
  \item So we want to \textbf{assess how well they do real scientific research}.
  \item To know, we need \textbf{benchmarks that give us information about that}.
\end{itemize}
\vfill
\begin{block}{Two papers, two benchmarks --- each evaluates a \emph{model $+$ a harness} in a container}
\begin{itemize}\footnotesize
  \item \textbf{GeneBench-Pro} \sm{(OpenAI)} --- \textbf{129 problems}, genomics. \gd{Deterministic Python grading.}
  \item \textbf{DrugDiscoveryBench} \sm{(Scale AI \& Phylo)} --- \textbf{82 tasks}, drug discovery. \rd{LLM-judge grading.}
\end{itemize}
\end{block}
\end{frame}

% ###########################################################################
\section{GeneBench-Pro}
% ###########################################################################

\begin{frame}{GeneBench-Pro --- the motivation}
\begin{block}{A gap in biology benchmarks}
Real scientific analysis is \textbf{far more complex} than what the benchmarks measure, and there are
\textbf{too few representative} ones. GBP exists to close that gap.
\end{block}
\bigskip
\begin{block}{Pointed at the right bottleneck}
The limiting factor is \textbf{no longer sample collection} but \textbf{downstream computation and
analysis}. So a good benchmark assesses the step \textbf{AI agents could most help with}.
\smallskip

\sm{``downstream computation and analysis, rather than data generation, have become the central
scientific bottleneck.'' --- GBP, p.\,2}
\end{block}
\end{frame}

\begin{frame}{GBP --- characteristics}
\begin{itemize}
  \item \textbf{129 problems}, 10 domains. \textbf{Fully simulated} --- causal structure known, nothing memorisable.
  \item Docker, \textbf{no internet}, a minimal prompt. Files are \textbf{deliberately messy} \sm{(sentinels, batch effects, missingness).}
\end{itemize}
\vfill
\begin{alertblock}{The unit to remember --- the \emph{decision point}}
A \textbf{fork where two reasonable paths give \emph{different numbers}} --- e.g. a \code{-999} column:
take its mean as-is, or recognise it as \emph{missing} and drop it first. \textbf{3--13 per problem} (median 6), chained.
\smallskip

GBP grades \textbf{only the final number} --- but a problem is built so the right number is reachable
\textbf{only by the right choice at every fork.} \sm{So one number tests judgement, not just arithmetic.}
\end{alertblock}
\end{frame}

\begin{frame}{GBP --- why build it synthetically}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{alertblock}{Two ways such benchmarks break}
\footnotesize
\textbf{Messy historical data} $\to$ many defensible paths $\to$ the score measures the \textbf{author's
preference}, not skill.
\medskip

\textbf{Numerically insensitive} $\to$ a \textbf{fundamental error} still passes.
\end{alertblock}
\end{column}
\begin{column}{0.5\textwidth}
\begin{exampleblock}{GBP's fix: \emph{simulate}}
\footnotesize
They know the true data-generating process --- so they can \textbf{compute} the answer (not assume it),
\textbf{prove by ablation} the naive path lands on a different number, and be sure \textbf{nothing was memorised.}
\end{exampleblock}
\end{column}
\end{columns}
\vfill
\centering\textbf{With real data you can only \emph{assume} a path is wrong. Simulated, you can \emph{prove} it.}
\end{frame}

\centering

\vfill
\begin{alertblock}{External review --- the human check on machine-made tasks}
\footnotesize
Synthetic data risks being \textbf{unrealistic} or \textbf{under-specified}. So \textbf{11 domain experts}
\sm{(grad students to professors)} reviewed 84 problems; \textbf{82 kept}. Each asked: \textbf{realistic?
is the answer \emph{uniquely} recoverable from the given data? are the methods appropriate?} Feedback drove redesigns.
\end{alertblock}
\end{frame}

\begin{frame}{GBP --- grading}
Grade against the \textbf{recoverable} estimate --- \emph{not} the hidden generating parameter:
\medskip
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{alertblock}{Naive (broken)}
\footnotesize
True knob $\beta=\mathbf{0.30}$; but a flawless analyst recovers $\hat\beta=\mathbf{0.283}$ from the
finite files. Grade against $0.30$: \rd{a perfect analysis is marked WRONG.}
\end{alertblock}
\end{column}
\begin{column}{0.5\textwidth}
\begin{exampleblock}{GBP (right)}
\footnotesize
Grade against \textbf{0.283} --- what a correct analysis can actually recover. \gd{Tests the analysis, not the dice.}
\end{exampleblock}
\end{column}
\end{columns}
\vfill
\centering\sm{Why they differ: the staged file is a \textbf{finite sample}, so noise moves the recoverable value off the knob --- like getting 6 heads in 10 flips of a fair coin.}
\smallskip

\textbf{Binary, deterministic Python --- no LLM judge.}\enspace\sm{Partial credit measures effort; binary measures usefulness.}
\end{frame}

\begin{frame}{GBP --- the results are brutal}
\centering\small
\begin{tabular}{@{}lrr@{}}
\toprule
\textbf{Configuration} & \textbf{Pass rate} & \textbf{Never solved \sm{(0/all tries)}} \\
\midrule
GPT-5.6 Sol Pro (Extended) --- \emph{best in paper} & \textbf{31.5\%} & \\
GPT-5.6 Sol (max) --- \emph{best mainline}          & \textbf{28.7\%} & \rd{45.7\%} \\
Claude Opus 4.8 (max) --- \emph{strongest non-GPT}  & 16.0\% & \\
\midrule
\multicolumn{3}{@{}l@{}}{\sm{Non-GPT range: \textbf{0.6\%} (MiniMax M2.7) to 16.0\% (Opus 4.8)}}\\
\bottomrule
\end{tabular}
\vfill
\begin{alertblock}{28.7\% is an average --- read the last column}
The best mainline model scores \rd{zero on 45.7\% of problems}, across all ten tries. \sm{Effort helps
then saturates: none$\to$max is $3.7 \to 28.7$, but high$\to$max is only $+1.9$.}
\end{alertblock}
\end{frame}

\begin{frame}{GBP --- limitations}
\begin{itemize}
  \item \rd{\textbf{OpenAI evaluates its own models.}} The leaderboard is topped by GPT-5.6, same lab ---
        the headline-model choice and framing are in-house.
  \medskip
  \item \rd{\textbf{No human baseline.}} 28.7\% has \textbf{no ceiling to read against} --- is an expert
        90\%? 60\%? The paper does not say.
\end{itemize}
\vfill
\centering\sm{So the numbers say how good these models are at \emph{these tasks} --- not how far from a scientist.}
\end{frame}

\section{DrugDiscoveryBench}
% ###########################################################################

\begin{frame}{DrugDiscoveryBench --- the motivation}
\why{The bigger goal: agents that can \emph{do science} --- form hypotheses, plan experiments, use domain tools, read the results.}
\begin{block}{Why \emph{drug discovery}, and why the \emph{early} part?}
Early drug discovery is a \textbf{narrow but especially testable} slice of that goal: it turns a hypothesis
into \textbf{evidence-gathering and analysis} over \textbf{structures, assays, patents, papers, and biological databases}.
\end{block}
\vfill
\begin{alertblock}{``Testable'' because every workflow demands three concrete things}
\centering
\gd{choose the right sources} \enspace$\cdot$\enspace \gd{keep the constraints across steps} \enspace$\cdot$\enspace \gd{return a \emph{verifiable} answer}
\smallskip

\raggedright\sm{All three can be checked --- and, as Part 1's failures showed, all three are exactly where agents slip.}
\end{alertblock}
\end{frame}

\begin{frame}{DDB --- what it measures}
\why{A drug is a small molecule that jams a \textbf{target} protein --- a key in a lock. DDB tests the \emph{early, computational} half of finding one.}
\begin{block}{The drug-discovery funnel}
\centering
\gd{target ID} $\to$ \gd{hit ID} $\to$ \gd{hit-to-lead} $\to$ \gd{lead opt.}
\;$\big|$\; \sm{candidate selection $\to$ preclinical $\to$ clinical}
\smallskip

\raggedright\footnotesize \gd{find the protein} $\to$ \gd{find molecules that stick} $\to$ \gd{keep the best}
$\to$ \gd{tune them}. \quad DDB grades these four and \textbf{stops at candidate selection}.
\end{block}
\vfill
\begin{alertblock}{Each task is a chain: \gd{retrieve} $\to$ \gd{parse} $\to$ \gd{filter} $\to$ \gd{compute} $\to$ one answer}
\textbf{3--12 dependent steps} (mean 6.3), and \textbf{a slip at any one sinks the whole answer.}
\sm{Just like GBP's cascading decision points --- which is why grading is all-or-nothing.}
\end{alertblock}
\end{frame}

\begin{frame}{DDB --- the tasks: real artifacts, one verifiable answer}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{itemize}\footnotesize
  \item \textbf{82 tasks}, written \emph{and} graded by domain experts.
  \item Each tied to \textbf{one real artifact} \sm{(a PDB structure, a patent, a paper, a database record)} that \textbf{already contains the answer.}
  \item Tagged by \textbf{skill} (7) $\times$ \textbf{funnel stage} (4) \sm{--- hit-to-lead densest (34 tasks), lead-opt thinnest (10).}
\end{itemize}
\smallskip
\centering\sm{\textbf{GBP simulates the truth; DDB looks it up.}}
\end{column}
\begin{column}{0.5\textwidth}
\begin{exampleblock}{One task, start to finish \sm{(lead optimisation)}}
\footnotesize
\emph{Compound B jams KRAS harder than A --- why?} Two \textbf{co-crystals} \sm{(the protein frozen with
each drug inside)}, \code{6USX}, \code{6UT0}:
\smallskip

\textbf{overlay} $\to$ find the \textbf{residue} \sm{(amino-acid bead)} that \textbf{moves} and touches B.
\quad\textbf{Answer: \code{THR58}.}
\end{exampleblock}
\smallskip
\centering\sm{Answers stay small: a count, a float, a molecule string (SMILES), a ranking.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{DDB --- the workbench (Scale-Biomni)}
\why{The agent is a \textbf{general coding agent} --- no drug-discovery training. DDB just hands it a biology toolbox.}
\begin{block}{What the agent is given}
\begin{itemize}\footnotesize
  \item \textbf{226 functions / 22 domains}, \textbf{imported like any Python library} \sm{(\code{query\_uniprot(...)}) --- no special wrapper between model and tools.}
  \item \textbf{76-file data lake} \sm{(reference tables on disk)} $+$ \textbf{117 packages}. \textbf{Internet on. 120 min/task.}
\end{itemize}
\end{block}
\vfill
\begin{alertblock}{Sealed vs open --- the contrast that will matter}
\centering \textbf{GBP:} no internet. \qquad \textbf{DDB:} internet $+$ a 226-function toolbox.
\smallskip

\sm{\emph{The interface is general; the work is biomedical.} And ``open'' means the agent can accidentally \textbf{look up the answer} (last limitation).}
\end{alertblock}
\end{frame}

\begin{frame}{DDB --- construction: the authoring pipeline}
\centering
\begin{tikzpicture}[font=\scriptsize,
  wf/.style={draw, rounded corners=2pt, fill=unipale, align=center, text width=1.72cm, minimum height=11mm, inner sep=1.5pt},
  ar/.style={-{Latex[length=1.6mm]}, thick, unigrey}]
\node[wf] (s1) {\textbf{1.} expert authors\\\sm{task + answer}\\\sm{+ playbook}};
\node[wf, right=3mm of s1] (s2) {\textbf{2.} second\\expert reviews};
\node[wf, right=3mm of s2] (s3) {\textbf{3.} BenchGuard\\\sm{auto: fidelity,}\\\sm{ambiguity}};
\node[wf, right=3mm of s3, fill=unired!8, draw=unired] (s4) {\textbf{4.} solvability\\\sm{playbook hint}\\\sm{$\to$ solvable?}};
\node[wf, right=3mm of s4] (s5) {\textbf{5.} SME fixes\\\rd{\sm{or DROPS}}};
\node[wf, right=3mm of s5, fill=unigreen!10] (s6) {\textbf{6.} independent\\QC sign-off};
\draw[ar] (s1)--(s2); \draw[ar] (s2)--(s3); \draw[ar] (s3)--(s4); \draw[ar] (s4)--(s5); \draw[ar] (s5)--(s6);
\end{tikzpicture}

\vfill
\begin{alertblock}{Gate 4 is the \emph{entry ticket} --- remember it for the results}
A task is kept only if an agent \textbf{can solve it once handed the expert's playbook} \sm{(the worked
recipe)}; the rest are \textbf{dropped}. \sm{Like a textbook that only keeps problems solvable \emph{with}
the worked solution --- so later, ``give them the playbook and they pass'' is partly baked in.}
\end{alertblock}
\end{frame}

\begin{frame}{DDB --- how an open answer gets graded}
\why{You can't \code{==} a paragraph. Each task ships with a \textbf{checklist}; an LLM ticks the boxes.}
\begin{block}{A \emph{weighted rubric} --- small checks, two kinds \sm{(KRAS task)}}
\footnotesize
\gd{outcome} \sm{--- scored:} ``answer is \code{THR58}'' \qquad
\textcolor{unigrey}{\textbf{process}} \sm{--- recorded, not scored:} ``fetched \code{6USX}$+$\code{6UT0}''
\smallskip

\centering\textbf{Pass $=$ 100\% of outcome checks.} \sm{A half-right answer, a timeout, or a refusal all score 0.}
\end{block}
\vfill
\begin{block}{The judge is an \textbf{LLM} (GPT-5.4), and it's \emph{strict}}
\footnotesize
Concrete checks, so it's \textbf{matching a claim to a key}, not essay-grading \sm{(3 LLM judges agree, $\kappa=1.0$).}
It forgives \textbf{last-digit rounding} on computed floats --- but \textbf{IDs, accession numbers, counts and
categories must match \emph{exactly}.}
\end{block}
\end{frame}

\begin{frame}{DDB --- results}
\centering\small
\begin{tabular}{@{}llr@{}}
\toprule
\textbf{Setting} & \textbf{Harness} & \textbf{Pass} \\
\midrule
GPT-5.5 (xhigh) --- \emph{best} & mini-SWE-agent & \gd{51.6\%} \\
Gemini 3.5 Flash (high)         & Gemini CLI     & 50.0\% \\
Claude Opus 4.8 (max)           & Claude Code    & 45.1\% \\
GPT-5.5 (xhigh)                 & Codex \emph{(native)} & 43.9\% \\
MiniMax M3 (xhigh) --- \emph{bottom} & mini-SWE-agent & \rd{23.2\%} \\
\bottomrule
\end{tabular}
\vfill
\begin{itemize}\footnotesize
  \item \textbf{Even the best solves just over half} (51.6\%) --- with tools, internet and 2 hours.
  \item \textbf{Think longer $\to$ score higher} \sm{(effort low$\to$ceiling: GPT-5.5 Codex $27.6\to39.8\to43.9$). One model generation $\approx$ 8--20 pts.}
  \item \textbf{The \emph{harness} matters as much as the model} \sm{--- swap GPT-5.5's wrapper, $43.9\to51.6$. ``Same chef, different kitchen.''}
\end{itemize}
\vfill
\centering\sm{Scores average \textbf{3 runs} per setting \sm{(not best-of-n)}. Coverage differs too: Opus refuses 5--6 published-data tasks per run.}
\end{frame}

\begin{frame}{DDB --- is the gap \emph{planning} or \emph{execution}?}
\why{Why do agents fail? Either they can't \emph{do} the steps, or they can't \emph{figure out} which steps. This experiment tells them apart.}
\begin{block}{Hand over the \emph{plan}, never the answer}
Give the agent the \textbf{playbook} \sm{(the recipe: which steps, which tools)} and see whether a failing
run is rescued.
\end{block}
\vfill
\centering\small
\begin{tabular}{@{}lr@{}}
\toprule
Best \emph{single} agent, unaided & 51.6\% \\
Solved by \emph{some} run \sm{(union of everyone's best --- an \emph{oracle})} & 76 / 82 \\
\rd{Solved once handed the plan} & \gd{80 / 82} \\
\bottomrule
\end{tabular}
\vfill
\begin{alertblock}{The conclusion}
Give the \textbf{plan} and the stragglers fall $\Rightarrow$ agents \textbf{can do the steps}; they just
can't \textbf{choose them}.
\medskip

\centering\textbf{The gap is unguided \emph{planning}, not execution.}
\medskip

\raggedright\sm{And when they fail, it's usually \textbf{dropping a constraint mid-run} \sm{(e.g. forgetting to filter for the disease)} or picking the \textbf{wrong data representation} --- exactly Part 1's Failure 2.}
\end{alertblock}
\end{frame}

\begin{frame}{DDB --- limitations}
\begin{itemize}\footnotesize
  \item \rd{\textbf{The judge was never checked against a human.}} Every score is one LLM's (GPT-5.4),
        ``validated'' only against \emph{two other LLMs}. Three LLMs agreeing may just mean they \textbf{share a
        blind spot} --- \sm{and the judge is GPT, same family as the winner.}
  \item \rd{\textbf{The ``54\% reasoning failure'' number is one run per task.}} A single noisy attempt each
        \sm{(the same task can go 1/10 then 6/10)} --- so it's ``54\% of the runs we looked at,'' not of failures.
  \item \rd{\textbf{The playbook result is partly circular.}} Solvable-with-playbook was the entry ticket
        (Gate 4); and \textbf{80/82} is best-of-many-runs, while the best \emph{single} agent is 51.6\%.
  \item \rd{\textbf{The internet leaked.}} Gemini CLI \textbf{found DDB's own results on the Scale blog and
        copied the answer} --- and it is \textbf{\#2 at 50.0\%}, never audited.
\end{itemize}
\end{frame}

\section{How these agents fail}
% ###########################################################################

\begin{frame}{Over half of failures are not coding failures}
\centering\small
\begin{tabular}{@{}lr@{}}
\toprule
\textbf{Failure mode \sm{(DDB, 226 failing runs)}} & \textbf{Share} \\
\midrule
\rd{Domain reasoning} \sm{--- wrong premise, though inputs and tools are correct} & \rd{54.0\%} \\
Derivation error \sm{--- right approach, wrong calculation} & 18.6\% \\
Retrieval \sm{--- wrong source, or never reads a file} & 16.4\% \\
Constraint \sm{--- violates an explicit instruction} & 7.5\% \\
Final-answer slip \sm{--- every step right, reports it wrong} & 3.5\% \\
\bottomrule
\end{tabular}
\vfill
\begin{alertblock}{The tools worked. The Python ran. The data was there.}
\textbf{Over half of failures are not coding failures.} \sm{The next five slides make each concrete --- and name the Part 2 mechanism it earns.}
\end{alertblock}
\end{frame}

\begin{frame}{Failure 1 --- it notices, and does nothing about it}
\begin{quote}\itshape
``models often complete substantial portions of the workflow but exhibit a consistent gap between
\textbf{noticing} and \textbf{acting}\ldots{}''
\end{quote}
\src{GeneBench-Pro, abstract, p.\,1}
\medskip
The model runs \code{df.describe()}, sees the sentinel values, may even comment on them --- \textbf{then
cleans the column and runs the analysis it had already decided on.}
\medskip

\centering\textbf{The observation never reaches the decision.}
\bridge{a \textbf{Findings Ledger} --- a noticed problem becomes an \emph{open obligation}, and the exit is blocked until it is discharged.}
\end{frame}

\begin{frame}{Failure 2 --- the question quietly degrades}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{block}{DDB --- the melanoma task}
\footnotesize
Mid-trajectory the qualifier \emph{``melanoma''} silently drops. The ranking becomes a gene-wide count;
\textbf{PTEN} wins on volume. \sm{(Truth: CDKN2A.)}
\end{block}
\end{column}
\begin{column}{0.5\textwidth}
\begin{block}{GBP --- a denominator trap}
\footnotesize
The estimand is over \textbf{the whole roster}; the agent standardises over \textbf{tested partners only}
--- a plausible, wrong answer.
\end{block}
\end{column}
\end{columns}
\medskip
\centering A constraint or population \textbf{silently drops}, and every step after is a correct analysis
\emph{of the wrong question}.
\bridge{a \textbf{Question Contract} --- write the estimand, population, units and constraints as state, and \emph{re-show it every turn}, so decay becomes visible.}
\end{frame}

\begin{frame}{Failure 3 --- the right method, on the wrong object}
The chemistry is \textbf{correct}, the method is \textbf{correct} --- run on the neutral \textbf{reference}
molecule instead of the \textbf{bound ligand}. Answer \textbf{0}, truth \textbf{$-$1}.
\medskip
\begin{alertblock}{And it self-checked}
The agent ran \textbf{three internal consistency checks}. All three agreed --- all three on the
\textbf{wrong molecule.}
\medskip

\centering\textbf{Internal consistency is not validity.}
\end{alertblock}
\bridge{a \textbf{fresh-context validator} --- a \emph{different} model that never sees the agent's reasoning, only the contract, the code, and what it printed.}
\end{frame}

\begin{frame}{Failure 4 --- its own code printed the right number}
\begin{quote}\itshape
``its own code printed the correct group-level count of 1, but the final tally used the atom-level
count of 2\ldots{} It reported 8 interactions instead of 7.''
\end{quote}
\src{DrugDiscoveryBench, p.\,27}
\medskip
The analysis, the code, and the printed output were all \textbf{right}. \rd{The number in the answer was not the number in the output.}
\bridge{\textbf{Evidence Grounding} --- regex every reported number against real stdout; reject the answer if a number was never printed. \emph{Fifteen lines.}}
\end{frame}

\begin{frame}{Failure 5 --- nobody checks the answer at the end}
\begin{quote}\itshape
``\textbf{The last chance to catch the slip is at the final answer}\ldots{} \textbf{None of the failing
models caught this.}''
\end{quote}
\src{DrugDiscoveryBench, p.\,13}
\medskip
Claude Opus answers \textbf{BRCA2} --- a \emph{breast}-cancer gene --- to a \textbf{melanoma} question.
Any expert would stop dead. \textbf{The agent submits it.}
\medskip

\centering\sm{One diagnosis, two labs: the bottleneck is \textbf{procedure}, not code, knowledge, or compute --- \textbf{and procedure is something an engineer can build.}}
\bridge{finishing is a \textbf{gated action} --- an LLM judge must \emph{approve} the draft before \code{submit\_answer} returns.}
\end{frame}

\section{The design}
% ###########################################################################

\begin{frame}{The assignment, and the thesis}
\begin{block}{Part 2, Option A}
A question in English $+$ a folder of data files. Explore, choose an approach, write and run code, check
intermediate results, return a \textbf{structured answer}.
\end{block}
\vfill
\centering\textbf{The base model already writes competent pandas. The work is making it \emph{keep the thread.}}
\smallskip

Part 1 says agents drop exactly \textbf{three} things --- so make each one state they cannot drop:
\medskip

\small
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{What gets dropped} & \textbf{The failure} & \textbf{The mechanism} \\
\midrule
\textbf{the question} & the melanoma qualifier evaporates & \textbf{Ledger 1 --- Question Contract} \\
\textbf{the finding}  & notices the sentinel, ignores it   & \textbf{Ledger 2 --- Findings Ledger} \\
\textbf{the number}   & printed 1, reported 2              & \textbf{Ledger 3 --- Evidence Grounding} \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}[fragile]{The base --- what an agent actually is}
\begin{enumerate}
  \item \textbf{The LLM is a pure function.} \code{messages} $\to$ \code{message}. No memory, no state.
  \item \textbf{Tool calling is not the model running code.} It emits a \emph{request}; \textbf{your program} executes it and appends the result.
  \item \textbf{The agent loop is a \code{while} loop.} \sm{(ReAct, Yao et al.)}
\end{enumerate}
\medskip
\begin{lstlisting}
while step < budget:
    msg = llm(transcript, tools)          # think
    if msg.tool_calls:
        result = execute(msg.tool_calls)  # act
        transcript.append(result)         # observe
    if submitted: break
\end{lstlisting}
\vfill
\centering\textbf{That is the whole agent --- $\sim$40 lines. Everything interesting surrounds the loop.}
\end{frame}

\begin{frame}{The architecture}
\centering

\smallskip
\centering\sm{The loop is unremarkable. \textbf{All of the design is the gated box on the right and the two ledgers pinned on top.}}

\begin{frame}{Three tools --- and why not more}
\begin{itemize}
  \item \code{run\_python(code)} --- a \textbf{persistent namespace}, like Jupyter cells. \sm{One general tool beats $N$ specific ones: anything pandas can do, the agent can do.}
  \item \code{note\_finding(...)} --- the ledger.
  \item \code{submit\_answer(...)} --- the \textbf{gated} exit.
\end{itemize}
\medskip
\sm{DDB gave its agents \textbf{226 functions} and still scored 51.6\% --- the models are \textbf{not failing for lack of affordances}.}
\vfill
\begin{block}{The single most important context rule}
\centering\textbf{The model never sees the data} --- only metadata, summaries, and the output of code it wrote.
\smallskip

\sm{Scale-invariant: a 200{,}000-row frame and a 200-row one cost the same tokens.}
\end{block}
\end{frame}

\begin{frame}{Addition 0 --- the detector}
A \textbf{deterministic profiler} runs at turn 0: shape, dtypes, nulls, cardinality --- \emph{and}
detectors for sentinels, duplicate ids, numeric-looking strings, group imbalance.
\medskip
\centering
\begin{tikzpicture}
\node[draw, rounded corners, fill=unired!8, draw=unired, text width=11.2cm, align=center, inner sep=6pt]
{What it finds is \textbf{not a helpful note} --- it is a \textbf{registered open obligation}, before turn one.\\[3pt]
\large\textbf{Information can be ignored. An obligation cannot.}};
\end{tikzpicture}
\vfill
\begin{block}{The division of labour the whole design rests on}
\centering\small
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{deterministic code} & finds the mechanical problems, and makes them \textbf{un-ignorable} \\
\textbf{the model}          & decides what they \textbf{mean} for \emph{this} question \\
\bottomrule
\end{tabular}
\end{block}
\end{frame}

\begin{frame}[fragile]{Ledger 1 --- the Question Contract}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{lstlisting}
class QuestionContract(BaseModel):
    estimand:   str   # the exact quantity
    population: str   # WHICH ROWS. denominator.
    units:      str   # percent? fraction? log?
    constraints: list[str]  # stated filters
    premises:    list[str]  # what it ASSUMES
    question_is_precise: bool   # REQUIRED
    ambiguities: list[str]
\end{lstlisting}
\begin{exampleblock}{Example --- a filled contract}
\scriptsize
\code{estimand}: mean biomarker change, wk 12\\
\code{population}: \textbf{arm A only}, completers\\
\code{premises}: [``arm A was randomised'']\\
\code{question\_is\_precise}: \gd{true}
\end{exampleblock}
\end{column}
\begin{column}{0.5\textwidth}
\begin{alertblock}{Two fields the evaluation demanded}
\footnotesize
\textbf{\code{premises}} --- a question can be \emph{wrong}. \emph{``Which of the \textbf{four} sites\ldots{}''}
--- there are \textbf{three}. Without it the agent answered anyway. \textbf{0/3 $\to$ 3/3.}
\medskip

\textbf{\code{question\_is\_precise}} --- a \emph{required boolean}. With an optional \code{ambiguities}
list, the agent left it empty \textbf{every time.}
\medskip

\centering\textbf{A field the model \emph{may} leave empty is a field it \emph{will} leave empty.}
\end{alertblock}
\smallskip
\centering\sm{Pinned every turn. Shown to the verifier at submit.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Ledger 2 --- the Findings Ledger}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{lstlisting}
note_finding(
  observation = "biomarker_x: 47 values "
                "of exactly -999.0",
  implication = "missing-data sentinels, "
                "not measurements",
  status      = "open",  # open|acted|dismissed
)
\end{lstlisting}
\begin{alertblock}{The rule that makes it bite}
\centering
\code{submit\_answer} is \textbf{hard-blocked} while \emph{any} finding is \code{open}.
\end{alertblock}
\smallskip
\sm{Close by \gd{act} (point at the code step) or \textbf{dismiss} (say why it doesn't matter). The
\code{implication} field is where the \emph{propagation} happens --- the step both papers watch it skip.}
\end{column}
\begin{column}{0.5\textwidth}
\begin{block}{Example --- a run's ledger}
\centering\scriptsize
\begin{tabular}{@{}p{0.20\linewidth}p{0.70\linewidth}@{}}
\toprule
\textbf{status} & \textbf{noticed $\to$ did} \\
\midrule
\gd{ACTED} & 88 \code{-999}s in a biomarker $\to$ excluded before any mean \\
\gd{ACTED} & 48 duplicate \code{patient\_id}s $\to$ kept the last row \\
\gd{ACTED} & \code{arm} imbalanced on \code{severity} $\to$ stratified \\
\textcolor{unigrey}{\textbf{DISMISS}} & batch B on a shifted scale $\to$ never enters a \emph{rate} \\
\bottomrule
\end{tabular}
\end{block}
\smallskip
\centering One is an \textbf{oracle}. The other is a \textbf{colleague.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Ledger 3 --- Evidence Grounding}
\begin{block}{The rule, in the system prompt, as an absolute}
\centering\textbf{Every number you report must have been printed by code you ran.}
\end{block}
\medskip
Enforced deterministically: regex the numbers, normalise \sm{(commas, \%, 4 sig figs, $\times100$/$\div100$)},
require containment in the concatenated stdout. \textbf{No LLM.}
\medskip
\begin{lstlisting}
REJECTED: the value 0.62 in your evidence never appeared in any executed
output. Compute and print it before citing it.
\end{lstlisting}
\vfill
\centering\textbf{An LLM reviewer \emph{might} catch ``printed 1, reported 2''. A regex catches it every time, for free.}
\end{frame}

\begin{frame}{The gated exit}
\code{submit\_answer} runs four gates \textbf{in order --- cheapest first}:
\medskip
\centering\small
\begin{tabular}{@{}clp{0.5\textwidth}@{}}
\toprule
& \textbf{Gate} & \textbf{Rejects when} \\
\midrule
1 & \textbf{schema}    & pydantic fails $\to$ the model self-repairs \\
2 & \textbf{ledger}    & \emph{any} finding is still \code{open} \\
3 & \textbf{grounding} & a reported number never appeared in real output \\
4 & \textbf{verifier}  & one LLM call --- \textbf{only if 1--3 passed} \\
\bottomrule
\end{tabular}
\vfill
\begin{alertblock}{The verifier's one critical detail}
Different model family, and \textbf{it does not see the agent's reasoning} --- only the contract, the
briefing, the code and its output, and the draft. \sm{Show it the chain of thought and it rubber-stamps.
\textbf{The fresh context \emph{is} the mechanism.}}
\end{alertblock}
\end{frame}

\begin{frame}{Everything, in one table}
\centering\small
\begin{tabular}{@{}p{0.28\textwidth}p{0.28\textwidth}p{0.36\textwidth}@{}}
\toprule
\textbf{The failure} & \textbf{From} & \textbf{What I add} \\
\midrule
Notices, does not act & \rd{GBP notice--act}; DDB \rd{54\%} & \textbf{Findings Ledger} --- an open obligation blocks the exit \\
\addlinespace
The question degrades & DDB constraint; melanoma & \textbf{Question Contract} --- pinned every turn \\
\addlinespace
Printed 1, reported 2 & DDB derivation $+$ slip & \textbf{Evidence Grounding} --- a regex on real stdout \\
\addlinespace
Nobody checks at the end & DDB final answer & \textbf{Fresh-context verifier} --- never sees the reasoning \\
\addlinespace
\emph{Never noticed at all} & \emph{\rd{neither paper measures this}} & \textbf{The detector} --- pre-seeds the ledger before turn 1 \\
\bottomrule
\end{tabular}
\vfill
\centering\sm{The last row is the one mechanism \emph{not} from the papers. \textbf{It will turn out to be the only one that matters.}}
\end{frame}

\begin{frame}{The benchmark}
\begin{block}{How I built it --- the simplest honest version}
\begin{enumerate}\footnotesize
  \item \textbf{Simulated two datasets} following GBP; \textbf{used a third, held-out} one.
  \item Asked \textbf{Claude to author 28 tasks} --- lookups, aggregations, and \emph{planted traps}.
  \item Grade with \textbf{binary, programmatic} metrics --- ground truth is a pandas callable only the grader runs.
\end{enumerate}
\end{block}
\medskip
\begin{itemize}\footnotesize
  \item \code{penguins.csv} --- real, clean. Guardrails should buy \emph{nothing}.
  \item \code{trial.csv} --- synthetic clinical, known data-generating process.
  \item \code{sales.csv} --- \rd{a held-out domain}, e-commerce. \textbf{Never designed against.}
\end{itemize}
\vfill
\centering\sm{GBP's third principle, reused: \emph{assert} the naive answer differs from the correct one by $\gg$ tolerance --- it \textbf{caught two of my own broken tasks.}}
\end{frame}

\begin{frame}{The ablation --- switch off one mechanism at a time}
\centering\small
\begin{tabular}{@{}lrr l@{}}
\toprule
\textbf{Remove this} & \textbf{Pass} & \textbf{$\Delta$} & \textbf{Verdict} \\
\midrule
\emph{(nothing --- the full agent)} & \textbf{87\%} & --- & \\
\midrule
\rd{the data briefing} \sm{$\leftarrow$ \textbf{the DETECTOR}} & \rd{60\%} & \rd{$-27\%$} & \rd{\textbf{HURTS}} \\
every guardrail at once & 67\% & $-20\%$ & \textbf{HURTS} \\
the Findings Ledger \sm{$\leftarrow$ \emph{centrepiece}} & 81\% & $-6\%$ & \textbf{HURTS} \\
\midrule
the fresh-context verifier & 87\% & $-0\%$ & \sm{no effect} \\
the numeric grounding gate & 87\% & $-0\%$ & \sm{no effect} \\
the Question Contract      & 88\% & $+1\%$ & \sm{no effect} \\
\bottomrule
\end{tabular}
\src{4{,}480 runs, 20/cell, $\sim$\$16. Paired hierarchical bootstrap (GBP's scheme).}
\vfill
\begin{alertblock}{Twenty lines of pandas beat every gate combined}
I built this around \textbf{gates}. The thing carrying it is the \textbf{detector} --- the mechanism I
spent the \emph{least} time on.
\end{alertblock}
\end{frame}

\begin{frame}{The conclusion}
\centering{\Large\textbf{A gate is only as good as the detector feeding it.}}
\smallskip

The gates weren't wrong --- they were \textbf{redundant.} Remove the detector and they have nothing to gate.
\vfill
\begin{alertblock}{Doesn't GBP say the opposite?}
GBP: \emph{frontier} models \emph{consistently notice} --- the headroom is in \emph{acting}. \textbf{But
that's a different model.} GBP runs GPT-5.6; my agent is \textbf{Qwen3-30B-A3B}, chosen because it is
weak. For a frontier model, noticing is the solved half; for a 30B model, \textbf{noticing is not reliable}
--- so the profiler substitutes for it.
\end{alertblock}
\vfill
\centering\textbf{The notice--act gap is a frontier problem. The \emph{notice} gap is a small-model problem.}\\
\sm{Which half of the scaffolding pays off depends on your base model --- an argument for owning an eval harness.}
\end{frame}

\begin{frame}{The failure the average hid}
\centering\small
\begin{tabular}{@{}lrr@{}}

\rd{\code{s4\_simpson\_sales}} --- Simpson's, \textbf{held-out} & \rd{35\%} & \rd{50\%} \\

\sm{25 easier tasks, 75--100\%} & \sm{carry the 87\%} & \\
\bottomrule
\end{tabular}

\begin{alertblock}{On the confounded comparison, my agent is worse than a coin flip}
It is \textbf{more likely to land on the naive answer (50--55\%) than the right one (35--40\%)}.
\medskip

\centering{\Large\textbf{An average is where a failure goes to hide.}}
\end{alertblock}
\smallskip
\centering\sm{And there is \textbf{no detector for confounding} --- so the roadmap is not more gates, it is \textbf{more detectors.}}
\end{frame}
